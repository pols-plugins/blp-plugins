cmake_minimum_required(VERSION 3.22)
project(BLPAutoPan VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_DIR "" CACHE PATH "Local JUCE source checkout")
if(JUCE_DIR)
  add_subdirectory("${JUCE_DIR}" juce)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
juce_add_plugin(BLPAutoPan COMPANY_NAME BLP PLUGIN_MANUFACTURER_CODE Blpa PLUGIN_CODE Apan
  PRODUCT_NAME "BLP Auto Pan" FORMATS VST3 VST3_CATEGORIES Fx Modulation Spatial)
target_sources(BLPAutoPan PRIVATE Plugin.cpp)
target_compile_definitions(BLPAutoPan PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(BLPAutoPan PRIVATE juce::juce_audio_processors juce::juce_recommended_config_flags)
