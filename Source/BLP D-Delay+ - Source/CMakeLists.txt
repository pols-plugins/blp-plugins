cmake_minimum_required(VERSION 3.22)
project(BLPDelay VERSION 1.0.0 LANGUAGES C CXX)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(BLPDelay COMPANY_NAME BLP PLUGIN_MANUFACTURER_CODE Blpa PLUGIN_CODE Ddpl FORMATS VST3 PRODUCT_NAME "BLP D-Delay+" VST3_CATEGORIES Fx Delay)
target_sources(BLPDelay PRIVATE Plugin.cpp)
target_compile_features(BLPDelay PRIVATE cxx_std_17)
target_compile_definitions(BLPDelay PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(BLPDelay PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
